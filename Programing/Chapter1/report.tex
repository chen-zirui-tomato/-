\section{部分代码}
\subsection{Bisection Method}
\begin{lstlisting}[language=C++]
for (i = 0; i < max_iter; i++) {
    h = h / 2;
    c = a + h;
    fc = F(c);
    if (fabs(fc) < eps || h < delta) {
        return c;
    } else if (fa * fc > 0) {
        a = c;
    }
}
\end{lstlisting}

\subsection{Newton's Method}
\begin{lstlisting}[language=C++]
for (i = 0; i < max_iter; i++) {
    double fx = F(x0);
    if (fabs(fx) < eps) {
        return x0;
    } else {
        x0 = x0 - fx / dfx;
        dfx = F.derivative(x0);
    }
}
\end{lstlisting}

\subsection{Secant Method}
\begin{lstlisting}[language=C++]
fa = F(a);
fb = F(b);
for (i = 0; i < max_iter; i++) {
    s = (b - a) / (fb - fa);
    a = b;
    fa = fb;
    b = b - s * fb;
    fb = F(b);
    if (fabs(b - a) < delta || fabs(fb) < eps) {
        return b;
    }
}
\end{lstlisting}
